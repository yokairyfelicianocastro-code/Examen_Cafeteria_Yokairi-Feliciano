        private const decimal RECARGO_GRANDE = 25m;
        private const decimal RECARGO_EXTRA = 30m;
        private const decimal EMPAQUE_LLEVAR = 20m;
        private const decimal PORCENTAJE_DESCUENTO = 0.10m;
        private const decimal PORCENTAJE_ITBIS = 0.18m;

        // Subtotal acumulado del pedido
        private decimal subtotal = 0m;

        private System.ComponentModel.IContainer components = null;

        // ---------- Agregar al pedido ----------
        private void btnAgregar_Click(object sender, EventArgs e)
        {
            if (txtProducto.Text.Trim() == "")
            {
                MessageBox.Show("Digite el nombre del producto.", "Cafeteria El Buen Sabor", MessageBoxButtons.OK, MessageBoxIcon.Warning);
                txtProducto.Focus();
                return;
            }

            decimal precioDigitado;
            if (!decimal.TryParse(txtPrecio.Text, out precioDigitado) || precioDigitado <= 0)
            {
                MessageBox.Show("Digite un precio valido mayor que cero.", "Cafeteria El Buen Sabor", MessageBoxButtons.OK, MessageBoxIcon.Warning);
                txtPrecio.Focus();
                return;
            }

            // Regla 1: precio unitario = precio digitado + 25 si es grande + 30 si lleva porcion extra
            decimal precioUnitario = precioDigitado;
            if (rbtGrande.Checked)
            {
                precioUnitario += RECARGO_GRANDE;
            }
            if (chkPorcionExtra.Checked)
            {
                precioUnitario += RECARGO_EXTRA;
            }

            // Regla 2: total de linea = precio unitario x cantidad
            decimal totalLinea = precioUnitario * nudCantidad.Value;
            subtotal += totalLinea;

            string tamano = rbtGrande.Checked ? "Grande" : "Normal";
            string productoConTamano = txtProducto.Text.Trim() + " (" + tamano + ")";

            lstPedido.Items.Add(
                ((int)nudCantidad.Value).ToString().PadRight(6) +
                productoConTamano.PadRight(24) +
                precioUnitario.ToString("0.00").PadLeft(10) +
                totalLinea.ToString("0.00").PadLeft(12));

            lblSubtotal.Text = "RD$ " + subtotal.ToString("0.00");

            // Reiniciar solo los controles del producto
            txtProducto.Clear();
            txtPrecio.Clear();
            nudCantidad.Value = 1;
            rbtNormal.Checked = true;
            chkPorcionExtra.Checked = false;
            txtProducto.Focus();
        }

        // ---------- Calcular total ----------
        private void btnCalcular_Click(object sender, EventArgs e)
        {
            if (lstPedido.Items.Count == 0)
            {
                MessageBox.Show("Debe agregar al menos un producto al pedido.", "Cafeteria El Buen Sabor", MessageBoxButtons.OK, MessageBoxIcon.Warning);
                return;
            }

            // Regla 3: descuento 10% si el cliente es frecuente
            decimal descuento = 0m;
            if (chkFrecuente.Checked)
            {
                descuento = subtotal * PORCENTAJE_DESCUENTO;
            }

            // Regla 4: empaque RD$20 una sola vez si es para llevar
            decimal empaque = 0m;
            if (rbtLlevar.Checked)
            {
                empaque = EMPAQUE_LLEVAR;
            }

            // Regla 5: base imponible = subtotal - descuento + empaque
            decimal baseImponible = subtotal - descuento + empaque;

            // Regla 6: ITBIS 18% de la base imponible, total = base + ITBIS
            decimal itbis = baseImponible * PORCENTAJE_ITBIS;
            decimal total = baseImponible + itbis;

            lblDescuento.Text = "RD$ " + descuento.ToString("0.00");
            lblEmpaque.Text = "RD$ " + empaque.ToString("0.00");
            lblItbis.Text = "RD$ " + itbis.ToString("0.00");
            lblTotal.Text = "RD$ " + total.ToString("0.00");
        }

        }
